##Master

CREATE USER 'repl'@'%' IDENTIFIED BY 'replpass';
GRANT REPLICATION SLAVE ON *.* TO 'repl'@'%';

docker compose up -d
docker ps

PS C:\Users\kunda\Sysdesign> docker exec sysdesign-mysql         mysql -uroot -prootpass -e "SELECT * FROM sysdesign.users;"

2       Komal



##Replica

docker exec sysdesign-mysql-replica mysql -uroot -prootpass -e "CHANGE REPLICATION SOURCE TO SOURCE_HOST='mysql', SOURCE_USER='repl', SOURCE_PASSWORD='replpass', SOURCE_AUTO_POSITION=1, GET_SOURCE_PUBLIC_KEY=1; START REPLICA; SET GLOBAL super_read_only=ON; SET GLOBAL read_only=ON;"

docker exec sysdesign-mysql-replica mysql -uroot -prootpass -e "SHOW REPLICA STATUS\G"
